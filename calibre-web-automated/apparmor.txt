#include <tunables/global>

# AppArmor profile for the Calibre-Web Automated add-on.
#
# The Supervisor loads this file because it sits next to config.yaml, and
# renames the profile to the add-on's slug. It replaces `apparmor: false`
# (unconfined), which the add-on needed only to bind-mount its storage.
#
# The base is Docker's default container profile (docker-default), which is
# what a container gets with no profile of its own. It adds a small set of
# mount rules: the add-on's init script bind-mounts the configured paths over
# the image's volumes and, when `smb_share` is set, mounts that share itself
# (rootfs/etc/s6-overlay/scripts/init-cwa-paths). Every other mount stays
# denied, just as under docker-default. It also denies the use of
# DAC_READ_SEARCH, which config.yaml grants only for mount.cifs (see below).
#
# The signal and ptrace rules carry no peer: the Supervisor renames only the
# `profile` line, so a peer naming this profile would not match the loaded
# name. Containers have their own PID namespace, so these processes cannot
# address anything outside it anyway.
profile calibre-web-automated flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  network,
  capability,
  # DAC_READ_SEARCH is granted (config.yaml) only so mount.cifs can put it
  # into its capability sets while dropping privileges; nothing here needs to
  # use it. Denying its use keeps open_by_handle_at -- the "Shocker" escape
  # from a container that holds it -- refused, as under Docker's defaults.
  deny capability dac_read_search,
  file,
  umount,
  signal (send, receive),
  ptrace (read, readby),

  # Bind mounts made by init-cwa-paths. `options=(move)` covers mount(8)
  # attaching the bind through the new mount API (open_tree + move_mount),
  # which util-linux uses when the kernel supports it.
  mount options=(rw, bind) -> /config/,
  mount options=(rw, move) -> /config/,
  mount options=(rw, bind) -> /calibre-library/,
  mount options=(rw, move) -> /calibre-library/,
  mount options=(rw, bind) -> /calibre-books/,
  mount options=(rw, move) -> /calibre-books/,
  mount options=(rw, bind) -> /cwa-book-ingest/,
  mount options=(rw, move) -> /cwa-book-ingest/,
  mount options=(rw, bind) -> /config/.config/calibre/plugins/,
  mount options=(rw, move) -> /config/.config/calibre/plugins/,

  # The `smb_share` mount, made by mount.cifs.
  mount fstype=cifs -> /mnt/smb/,
  mount options=(rw, move) -> /mnt/smb/,

  # From docker-default.
  deny @{PROC}/* w,   # deny write for all files directly in /proc (not in a subdir)
  # deny write to files not in /proc/<number>/** or /proc/sys/**
  deny @{PROC}/{[^1-9],[^1-9][^0-9],[^1-9s][^0-9y][^0-9s],[^1-9][^0-9][^0-9][^0-9/]*}/** w,
  deny @{PROC}/sys/[^k]** w,  # deny /proc/sys except /proc/sys/k* (effectively /proc/sys/kernel)
  deny @{PROC}/sys/kernel/{?,??,[^s][^h][^m]**} w,  # deny everything except shm* in /proc/sys/kernel/
  deny @{PROC}/sysrq-trigger rwklx,
  deny @{PROC}/kcore rwklx,

  deny /sys/[^f]*/** wklx,
  deny /sys/f[^s]*/** wklx,
  deny /sys/fs/[^c]*/** wklx,
  deny /sys/fs/c[^g]*/** wklx,
  deny /sys/fs/cg[^r]*/** wklx,
  deny /sys/firmware/** rwklx,
  deny /sys/devices/virtual/powercap/** rwklx,
  deny /sys/kernel/security/** rwklx,
}
